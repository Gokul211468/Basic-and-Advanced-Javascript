Task 1: Convert .then to async/await

What to do: Take this chain: fetchUser(7).then(u => fetchOrders(u.id)).then(orders => console.log(orders.length)).catch(e => console.error(e)). Mock fetchUser and fetchOrders with setTimeout that resolve in 300ms. Rewrite the chain as a single async function showOrders(id) using await and try/catch. Call it with showOrders(7).

function fetchUser(id) {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            resolve({
                id: id,
                name: "Gokul"
            });
        }, 300);
    });
}

function fetchOrders(userId) {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            resolve([
                "order1",
                "order2",
                "order3"
            ]);
        }, 300);
    });
}

// Original .then chain
fetchUser(7)
  .then(u => fetchOrders(u.id))
  .then(orders => console.log(orders.length))
  .catch(e => console.error(e))

// Converted to async/await
async function usingAsync(id) {
    try{
        const user = await fetchUser(id)
        const orders = await fetchOrders(user.id)
        console.log(orders.length)  // Output: 3
    }
    catch(error){
        console.log(error)
    }
}

usingAsync(7)

/*
Key insights:
1. async/await makes nested operations read like synchronous code
2. try/catch replaces .catch() for error handling
3. Each await pauses until the Promise resolves
4. Much more readable than .then() chains
*/

Task 2: Sequential vs Parallel Timing

What to do: Take a fetchPrice(id) that takes 500ms to resolve to { id, price: 100 }. Write slow() that fetches 3 products with separate awaits. Time it. Write fast() that uses Promise.all. Time it. Confirm slow ≈ 1500ms and fast ≈ 500ms.

function fetchPrice(id){
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            resolve({
                id: id,
                price: 100
            });
        }, 500);
    })
}

// SEQUENTIAL - Each await waits for previous to complete
async function slow(){
    const start = Date.now();
    const p1 = await fetchPrice(1);    // Wait 500ms
    const p2 = await fetchPrice(2);    // Wait another 500ms
    const p3 = await fetchPrice(3);    // Wait another 500ms
    console.log(`slow time is ${Date.now() - start} ms`)  // ~1500ms
}

// PARALLEL - All requests start immediately
async function fast(){
    const start = Date.now();
    const results = await Promise.all([
        fetchPrice(1), 
        fetchPrice(2), 
        fetchPrice(3)
    ]);
    console.log(`fast time is ${Date.now() - start} ms`)  // ~500ms
}

slow();   // ~1500ms - sequential execution
fast();   // ~500ms - parallel execution

/*
Critical insight:
- Sequential: Total time = sum of all operations (500 + 500 + 500 = 1500ms)
- Parallel: Total time = longest single operation (~500ms)
- Use Promise.all when operations are independent and can run simultaneously
*/

Task 3: Fix the forEach Trap

What to do: Take an array const ids = [1, 2, 3] and a fetchPrice(id) (500ms each, returns { id, price: 100 }). Try to write a function that uses ids.forEach(async (id) => ...) to log each price. Time it. Note that the function returns BEFORE any price logs. Fix it two ways: (1) for...of, (2) Promise.all + map.

const ids = [1, 2, 3];
function fetchProduct(id) {
    return new Promise((res) => setTimeout(() => res({ id, price: 100 }), 500));
}

// ❌ THE TRAP - forEach doesn't wait for async operations
console.log("=== forEach Trap Demo ===");
console.log("start");
ids.forEach(async (id) => {
    const p = await fetchProduct(id);
    console.log("got price:", p.price);
});
console.log("end");  // This logs IMMEDIATELY, before any "got price"

/*
Output order:
start
end
got price: 100
got price: 100  
got price: 100

Problem: forEach doesn't understand Promises!
*/

// ✅ FIX 1 — for...of (sequential execution)
async function sequential() {
    console.log("=== Sequential Fix ===");
    console.log("start");
    for (const id of ids) {
      const p = await fetchProduct(id);
      console.log("got", p);
    }
    console.log("end");                     // logs AFTER all "got"
}

// ✅ FIX 2 — Promise.all + map (parallel execution, preferred)
async function parallel() {
    console.log("=== Parallel Fix ===");
    console.log("start");
    const promises = ids.map(id => fetchProduct(id));  // Create array of Promises
    const products = await Promise.all(promises);      // Wait for all to complete
    products.forEach(p => console.log("got", p));
    console.log("end");
}

sequential();  // Takes ~1500ms (500ms × 3, sequential)
parallel();    // Takes ~500ms (all run in parallel)

/*
Key lessons:
1. Never use forEach with async/await - it doesn't wait!
2. Use for...of for sequential async operations
3. Use Promise.all + map for parallel async operations
4. map creates Promises, Promise.all waits for them
*/

Bonus: Retry with async/await

What to do: Write retry(fn, attempts) that calls async fn() and retries up to attempts times if it rejects. After all attempts fail, throw the last error. Test with a flaky() function that randomly resolves/rejects.

// Simulates a flaky network request
function flaky() {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            const success = Math.random() > 0.5;  // 50% success rate

            if (success) {
                resolve("Success!");
            } else {
                reject(new Error("Random failure"));
            }
        }, 300);
    });
}

// Retry mechanism with exponential backoff-like logging
async function retry(fn, attempts) {
    let lastError;

    for (let i = 1; i <= attempts; i++) {
        try {
            console.log(`Attempt ${i}/${attempts}`);
            
            const result = await fn();  // Try the function
            
            console.log(`✅ Success on attempt ${i}!`);
            return result;

        } catch (error) {
            console.log(`❌ Failed attempt ${i}: ${error.message}`);
            lastError = error;
            
            // Optional: Add delay between retries
            if (i < attempts) {
                console.log("Retrying in 100ms...");
                await new Promise(resolve => setTimeout(resolve, 100));
            }
        }
    }

    // All attempts failed
    console.log(`💥 All ${attempts} attempts failed`);
    throw lastError;
}

// Test the retry mechanism
retry(flaky, 3)
    .then((result) => {
        console.log("Final result:", result);
    })
    .catch((error) => {
        console.log(`Final error: ${error.message}`);
    });

/*
Possible outputs:

Success case:
Attempt 1/3
❌ Failed attempt 1: Random failure  
Retrying in 100ms...
Attempt 2/3
✅ Success on attempt 2!
Final result: Success!

Failure case:
Attempt 1/3
❌ Failed attempt 1: Random failure
Retrying in 100ms...
Attempt 2/3  
❌ Failed attempt 2: Random failure
Retrying in 100ms...
Attempt 3/3
❌ Failed attempt 3: Random failure
💥 All 3 attempts failed
Final error: Random failure

Real-world usage: API calls, database connections, file operations
*/
