Exercise 1: Three-level Chain with Object.create

What to do: Use Object.create to build a 3-level chain: tool → vehicle → car. Each level adds one method. Confirm a car instance can call all three methods.

const tool = {
    repair() {
      console.log("Repairing...");
    },
  };
  
  const vehicle = Object.create(tool);
  
  vehicle.start = function () {
    console.log("Starting...");
  };
  
  const car = Object.create(vehicle);
  
  car.drive = function () {
    console.log("Driving...");
  };
  
  car.drive();   // Driving...  (own method)
  car.start();   // Starting... (from vehicle prototype) 
  car.repair();  // Repairing... (from tool prototype)

  /*
  Prototype chain:
  car → vehicle → tool → Object.prototype → null
  
  Each level adds functionality:
  - tool: repair capability
  - vehicle: start capability + repair (inherited)
  - car: drive capability + start + repair (inherited)
  
  This demonstrates how Object.create builds inheritance chains
  */

Exercise 2: Shape/Circle Constructor Inheritance

What to do: Build a Shape(name) constructor with a describe() method on the prototype. Then build Circle(name, radius) that inherits from Shape and adds an area() method. Test with a Circle of name "C1" and radius 5.

function Shape(name) {
    this.name = name;
  }
  
  Shape.prototype.describe = function () {
    console.log("Shape name is " + this.name);
  };
  
  function Circle(name, radius) {
    Shape.call(this, name);  // Call parent constructor
    this.radius = radius;
  }
  
  // Set up inheritance
  Circle.prototype = Object.create(Shape.prototype);
  Circle.prototype.constructor = Circle;
  
  Circle.prototype.area = function () {
    console.log(Math.PI * this.radius * this.radius);
  };
  
  const c1 = new Circle("C1", 5);
  
  c1.describe(); // "Shape name is C1" (inherited from Shape)
  c1.area();     // 78.53981633974483 (own to Circle)

  /*
  Inheritance pattern:
  1. Shape.call(this, name) - borrows Shape constructor
  2. Circle.prototype = Object.create(Shape.prototype) - sets up prototype chain
  3. Circle.prototype.constructor = Circle - maintains constructor reference
  4. Circle instances can access both Shape and Circle methods
  
  This is the classical inheritance pattern in pre-ES6 JavaScript
  */

Exercise 3: Inspect Built-in Prototypes

What to do: Take any built-in prototype (e.g. String.prototype) and inspect it in the console. Find five methods you didn't know about. Document one with a code example.

console.log(String.prototype);

// Some less commonly noticed String methods:
//
// 1. at() - access character at index (supports negative)
// 2. padStart() - pad string to length from start
// 3. padEnd() - pad string to length from end
// 4. matchAll() - find all matches of a regex
// 5. normalize() - normalize Unicode representation

// Example using at()
const str = "JavaScript";

console.log(str.at(2));   // "v" - character at index 2
console.log(str.at(-1));  // "t" - last character
console.log(str.at(-2));  // "p" - second to last

// Why at() is useful:
// - Supports negative indices (like Python)
// - More intuitive than str[str.length - 1] for last character
// - Returns undefined for out-of-bounds (rather than empty string)

/*
Built-in prototypes are treasure troves of functionality:
- String.prototype: 50+ methods for text manipulation
- Array.prototype: 40+ methods for array operations  
- Object.prototype: Base methods all objects inherit
- Number.prototype: Methods for number formatting/conversion
- Date.prototype: Methods for date/time manipulation

Exploring these helps you write more idiomatic JavaScript!
*/

Exercise 4: chainOf(obj) Function

What to do: Write a function chainOf(obj) that returns an array of every prototype in obj's chain, ending at the prototype just before null.

function chainOf(obj) {
    const chain = [];
  
    let current = Object.getPrototypeOf(obj);
  
    while (current !== null) {
      chain.push(current);
      current = Object.getPrototypeOf(current);
    }
  
    return chain;
  }
  
  const arr = [1, 2, 3];
  
  console.log(chainOf(arr));
  
  /*
  Output:
  [
    Array.prototype,    // Has push, pop, forEach, etc.
    Object.prototype    // Has toString, hasOwnProperty, etc.
  ]
  */

  // Test with different objects
  const obj = {};
  console.log(chainOf(obj));        // [Object.prototype]
  
  const date = new Date();
  console.log(chainOf(date));       // [Date.prototype, Object.prototype]
  
  function MyClass() {}
  const instance = new MyClass();
  console.log(chainOf(instance));   // [MyClass.prototype, Object.prototype]

  /*
  This function demonstrates:
  1. How to traverse the prototype chain programmatically
  2. The while loop pattern for chain walking
  3. Different objects have different chain lengths
  4. All chains end at Object.prototype (unless specifically created with null)
  
  Real-world usage: debugging inheritance issues, understanding method resolution
  */
