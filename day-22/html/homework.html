Exercise 1: Tag Counter with Map

What to do: Build a "tag counter": given an array of posts each with a .tags array, return a Map of tag → count. Use a Map (not an Object) and demonstrate how it's better for this use case.

// Sample data structure
const posts = [
    {
      title: "Getting Started with React",
      tags: ["js", "react", "frontend"]
    },
    {
      title: "Node.js Best Practices", 
      tags: ["js", "node", "backend"]
    },
    {
      title: "CSS Grid Layout",
      tags: ["css", "frontend", "layout"]
    },
    {
      title: "React Hooks Deep Dive",
      tags: ["js", "react", "hooks"]
    }
];

// Tag counter implementation using Map
function countTags(posts) {
    const tagMap = new Map();
    
    for (const post of posts) {
        for (const tag of post.tags) {
            // Use Map's get with fallback pattern
            tagMap.set(tag, (tagMap.get(tag) || 0) + 1);
        }
    }
    
    return tagMap;
}

const tagCounts = countTags(posts);

console.log("Tag frequency analysis:");
console.log(tagCounts);
// Map(6) { 
//   'js' => 3, 'react' => 2, 'frontend' => 2, 
//   'node' => 1, 'backend' => 1, 'css' => 1, 'layout' => 1, 'hooks' => 1 
// }

// Convert to sorted array for display
const sortedTags = [...tagCounts]
    .sort(([,a], [,b]) => b - a)  // Sort by count descending
    .map(([tag, count]) => `${tag}: ${count}`);

console.log("Most popular tags:");
sortedTags.forEach(tag => console.log(`  ${tag}`));

// Why Map is better than Object for this:
console.log("\nMap advantages:");
console.log("Direct size:", tagCounts.size);  // O(1) vs Object.keys(obj).length
console.log("Easy iteration:", [...tagCounts.keys()]);
console.log("No prototype pollution risk");



function countTagsAlt(posts) {
    const tagMap = new Map();
    
    posts.forEach(post => 
        post.tags.forEach(tag => 
            tagMap.set(tag, (tagMap.get(tag) || 0) + 1)
        )
    );
    
    return tagMap;
}

Exercise 2: Set Operations Utility Library

What to do: Build a union(a, b), intersection(a, b), difference(a, b) set of helpers from scratch using Set + spread. Test with overlapping sets and demonstrate mathematical set properties.

// Set operations utility library
class SetMath {
    // Union: A ∪ B (all elements from both sets)
    static union(a, b) {
        return new Set([...a, ...b]);
    }
    
    // Intersection: A ∩ B (elements present in both sets)
    static intersection(a, b) {
        return new Set([...a].filter(value => b.has(value)));
    }
    
    // Difference: A - B (elements in A but not B)  
    static difference(a, b) {
        return new Set([...a].filter(value => !b.has(value)));
    }
    
    // Symmetric Difference: A ⊕ B (elements in either A or B, but not both)
    static symmetricDifference(a, b) {
        const diff1 = SetMath.difference(a, b);
        const diff2 = SetMath.difference(b, a);
        return SetMath.union(diff1, diff2);
    }
    
    // Subset: A ⊆ B (all elements of A are in B)
    static isSubset(a, b) {
        return [...a].every(value => b.has(value));
    }
    
    // Superset: A ⊇ B (A contains all elements of B)  
    static isSuperset(a, b) {
        return SetMath.isSubset(b, a);
    }
    
    // Disjoint: A ∩ B = ∅ (no common elements)
    static areDisjoint(a, b) {
        return SetMath.intersection(a, b).size === 0;
    }
}

// Test sets
const frontend = new Set(["html", "css", "javascript", "react"]);
const backend = new Set(["javascript", "node", "express", "mongodb"]);
const mobile = new Set(["react", "swift", "kotlin"]);

console.log("Set A (Frontend):", [...frontend]);
console.log("Set B (Backend):", [...backend]);
console.log("Set C (Mobile):", [...mobile]);

// Mathematical operations
console.log("\n=== Set Operations ===");

const unionAB = SetMath.union(frontend, backend);
console.log("A ∪ B (Union):", [...unionAB]);
// All technologies: html, css, javascript, react, node, express, mongodb

const intersectionAB = SetMath.intersection(frontend, backend);
console.log("A ∩ B (Intersection):", [...intersectionAB]);
// Common technologies: javascript

const differenceAB = SetMath.difference(frontend, backend);
console.log("A - B (Difference):", [...differenceAB]);
// Frontend-only: html, css, react

const symmetricDiffAB = SetMath.symmetricDifference(frontend, backend);
console.log("A ⊕ B (Symmetric Diff):", [...symmetricDiffAB]);
// Either frontend or backend, not both: html, css, react, node, express, mongodb

// Set relationships
console.log("\n=== Set Relationships ===");
const jsOnly = new Set(["javascript"]);
console.log("Is {javascript} ⊆ Frontend?", SetMath.isSubset(jsOnly, frontend));  // true
console.log("Is Frontend ⊇ {javascript}?", SetMath.isSuperset(frontend, jsOnly));  // true

Exercise 3: Advanced Map Sorting and Transformation

What to do: Convert a Map of { name → age } to an array of [name, age] pairs, sorted by age. Use destructuring and demonstrate multiple sorting approaches.

// Sample data: Map of person names to ages
const ages = new Map([
    ["Priya", 24],
    ["Aarav", 30], 
    ["Riya", 21],
    ["Anaya", 27],
    ["Karan", 35],
    ["Diya", 19]
]);

console.log("Original Map:");
console.log(ages);

// Method 1: Sort by age (ascending)
const sortedByAge = [...ages].sort(([, ageA], [, ageB]) => ageA - ageB);

console.log("\nSorted by age (youngest first):");
sortedByAge.forEach(([name, age]) => {
    console.log(`${name}: ${age} years old`);
});

// Method 2: Sort by name (alphabetical)
const sortedByName = [...ages].sort(([nameA], [nameB]) => nameA.localeCompare(nameB));

console.log("\nSorted by name (alphabetical):");
sortedByName.forEach(([name, age]) => {
    console.log(`${name}: ${age} years old`);
});

// Method 3: Advanced filtering and sorting
const adults = [...ages]
    .filter(([, age]) => age >= 21)  // Filter adults only
    .sort(([, ageA], [, ageB]) => ageB - ageA);  // Sort oldest first

console.log("\nAdults only (oldest first):");
adults.forEach(([name, age]) => {
    console.log(`${name}: ${age} years old`);
});

// Method 4: Transform to objects with additional properties
const enrichedData = [...ages]
    .map(([name, age]) => ({
        name,
        age, 
        category: age < 21 ? 'Teen' : age < 30 ? 'Young Adult' : 'Adult',
        initials: name.split('').filter(char => char === char.toUpperCase()).join('')
    }))
    .sort((a, b) => a.age - b.age);

console.log("\nEnriched and categorized data:");
enrichedData.forEach(person => {
    console.log(`${person.initials} - ${person.name} (${person.age}): ${person.category}`);
});

// Method 5: Group by age ranges using Map
function groupByAgeRange(peopleMap) {
    const ranges = new Map([
        ['teens', []],
        ['twenties', []],  
        ['thirties', []]
    ]);
    
    for (const [name, age] of peopleMap) {
        if (age < 20) {
            ranges.get('teens').push({ name, age });
        } else if (age < 30) {
            ranges.get('twenties').push({ name, age });
        } else {
            ranges.get('thirties').push({ name, age });
        }
    }
    
    // Sort within each group
    ranges.forEach(group => group.sort((a, b) => a.age - b.age));
    
    return ranges;
}

const groupedAges = groupByAgeRange(ages);

console.log("\nGrouped by age ranges:");
groupedAges.forEach((people, range) => {
    console.log(`${range.toUpperCase()}:`);
    people.forEach(({ name, age }) => {
        console.log(`  ${name} (${age})`);
    });
});

// Method 6: Statistical analysis
const ageStats = {
    count: ages.size,
    ages: [...ages.values()],
    get average() {
        return this.ages.reduce((sum, age) => sum + age, 0) / this.count;
    },
    get median() {
        const sorted = [...this.ages].sort((a, b) => a - b);
        const mid = Math.floor(sorted.length / 2);
        return sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
    },
    get min() {
        return Math.min(...this.ages);
    },
    get max() {
        return Math.max(...this.ages);
    }
};

console.log("\nAge Statistics:");
console.log(`Count: ${ageStats.count}`);
console.log(`Average: ${ageStats.average.toFixed(1)} years`);
console.log(`Median: ${ageStats.median} years`);
console.log(`Range: ${ageStats.min} - ${ageStats.max} years`);
